CREATE OR REPLACE PROCEDURE dw.load(p_target VARCHAR)
RETURNS STRING
LANGUAGE SQL
EXECUTE AS CALLER
AS $$
DECLARE
  n INT := 0;
  c1 CURSOR FOR SELECT id FROM src;
BEGIN
  SELECT COUNT(*) INTO :n FROM src;
  OPEN c1;
  FETCH c1 INTO :n;
  CLOSE c1;
  IF n > 0 THEN
    MERGE INTO tgt t USING src s ON t.id = s.id
    WHEN MATCHED AND s.upd = TRUE THEN UPDATE SET t.v = s.v
    WHEN NOT MATCHED THEN INSERT (id, v) VALUES (s.id, s.v);
  ELSEIF n = 0 THEN
    RETURN 'empty';
  ELSE
    RAISE;
  END IF;
EXCEPTION
  WHEN OTHER THEN RAISE;
END;
$$;

RETURNS VARCHAR
LANGUAGE SQL
AS $$
DECLARE
  i INT;
BEGIN
  i := 0;
  WHILE i < 10 DO
    i := i + 1;
    IF i = 3 THEN
      CONTINUE;
    END IF;
  END WHILE;
  FOR j IN 1 TO 5 DO
    IF j = 4 THEN
      BREAK;
    END IF;
  END FOR;
  FOR rec IN (SELECT id FROM t) DO
    i := rec.id;
  END FOR;
  RETURN i;
END;

  FOR row IN res DO

  RAISE my_exception;
EXCEPTION
  WHEN my_exception THEN
